   409→// ─── Particle Background Canvas ─────────────────────────────────────────────
   410→
   411→function ParticleBackground() {
   412→  const canvasRef = useRef<HTMLCanvasElement>(null);
   413→
   414→  useEffect(() => {
   415→    const canvas = canvasRef.current;
   416→    if (!canvas) return;
   417→    const ctx = canvas.getContext('2d');
   418→    if (!ctx) return;
   419→
   420→    let animId: number;
   421→    let particles: { x: number; y: number; vx: number; vy: number; size: number; opacity: number; color: string }[] = [];
   422→    let gridOffset = 0;
   423→
   424→    const resize = () => {
   425→      canvas.width = window.innerWidth;
   426→      canvas.height = window.innerHeight;
   427→    };
   428→    resize();
   429→    window.addEventListener('resize', resize);
   430→
   431→    // Create particles
   432→    for (let i = 0; i < 60; i++) {
   433→      particles.push({
   434→        x: Math.random() * canvas.width,
   435→        y: Math.random() * canvas.height,
   436→        vx: (Math.random() - 0.5) * 0.3,
   437→        vy: -Math.random() * 0.5 - 0.1,
   438→        size: Math.random() * 2 + 0.5,
   439→        opacity: Math.random() * 0.5 + 0.1,
   440→        color: Math.random() > 0.5 ? '#00f0ff' : '#10b981',
   441→      });
   442→    }
   443→
   444→    const drawGrid = (time: number) => {
   445→      const gridSpacing = 60;
   446→      const perspective = 0.003;
   447→      const pulse = Math.sin(time * 0.001) * 0.15 + 0.2;
   448→      gridOffset = (gridOffset + 0.15) % gridSpacing;
   449→
   450→      ctx.strokeStyle = `rgba(0, 240, 255, ${pulse * 0.3})`;
   451→      ctx.lineWidth = 0.5;
   452→      ctx.beginPath();
   453→
   454→      // Horizontal lines (perspective grid)
   455→      for (let y = canvas.height * 0.6; y < canvas.height; y += gridSpacing * 0.6) {
   456→        const factor = (y - canvas.height * 0.5) / (canvas.height * 0.5);
   457→        ctx.globalAlpha = factor * pulse;
   458→        ctx.moveTo(0, y);
   459→        ctx.lineTo(canvas.width, y);
   460→      }
   461→      ctx.stroke();
   462→
   463→      // Vertical lines converging to horizon
   464→      ctx.beginPath();
   465→      const vanishX = canvas.width / 2;
   466→      const vanishY = canvas.height * 0.45;
   467→      for (let i = -15; i <= 15; i++) {
   468→        const x = vanishX + i * gridSpacing * 1.5;
   469→        ctx.globalAlpha = pulse * 0.25;
   470→        ctx.moveTo(vanishX, vanishY);
   471→        ctx.lineTo(x + (x - vanishX) * 3, canvas.height);
   472→      }
   473→      ctx.stroke();
   474→      ctx.globalAlpha = 1;
   475→    };
   476→
   477→    const animate = (time: number) => {
   478→      ctx.clearRect(0, 0, canvas.width, canvas.height);
   479→
   480→      // Draw perspective grid
   481→      drawGrid(time);
   482→
   483→      // Draw particles
   484→      particles.forEach((p) => {
   485→        p.x += p.vx;
   486→        p.y += p.vy;
   487→
   488→        if (p.y < -10) { p.y = canvas.height + 10; p.x = Math.random() * canvas.width; }
   489→        if (p.x < -10) p.x = canvas.width + 10;
   490→        if (p.x > canvas.width + 10) p.x = -10;
   491→
   492→        ctx.beginPath();
   493→        ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
   494→        ctx.fillStyle = p.color;
   495→        ctx.globalAlpha = p.opacity;
   496→        ctx.fill();
   497→
   498→        // Glow
   499→        ctx.beginPath();
   500→        ctx.arc(p.x, p.y, p.size * 3, 0, Math.PI * 2);
   501→        const grad = ctx.createRadialGradient(p.x, p.y, 0, p.x, p.y, p.size * 3);
   502→        grad.addColorStop(0, p.color);
   503→        grad.addColorStop(1, 'transparent');
   504→        ctx.fillStyle = grad;
   505→        ctx.globalAlpha = p.opacity * 0.3;
   506→        ctx.fill();
   507→      });
   508→      ctx.globalAlpha = 1;
   509→
   510→      animId = requestAnimationFrame(animate);
   511→    };
   512→
   513→    animId = requestAnimationFrame(animate);
   514→
   515→    return () => {
