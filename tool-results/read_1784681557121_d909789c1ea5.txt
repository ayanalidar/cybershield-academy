    60→
    61→function useTypingEffect(phrases: string[], typingSpeed = 80, deletingSpeed = 40, pauseTime = 2000) {
    62→  const [displayText, setDisplayText] = useState('');
    63→  const [phraseIndex, setPhraseIndex] = useState(0);
    64→  const [isDeleting, setIsDeleting] = useState(false);
    65→
    66→  useEffect(() => {
    67→    const currentPhrase = phrases[phraseIndex];
    68→    let timeout: NodeJS.Timeout;
    69→
    70→    if (!isDeleting && displayText === currentPhrase) {
    71→      timeout = setTimeout(() => setIsDeleting(true), pauseTime);
    72→    } else if (isDeleting && displayText === '') {
    73→      timeout = setTimeout(() => {
    74→        setIsDeleting(false);
    75→        setPhraseIndex((prev) => (prev + 1) % phrases.length);
    76→      }, 0);
    77→    } else {
    78→      timeout = setTimeout(
    79→        () => {
    80→          setDisplayText(
    81→            isDeleting
    82→              ? currentPhrase.substring(0, displayText.length - 1)
    83→              : currentPhrase.substring(0, displayText.length + 1),
    84→          );
    85→        },
    86→        isDeleting ? deletingSpeed : typingSpeed,
    87→      );
    88→    }
    89→
    90→    return () => clearTimeout(timeout);
    91→  }, [displayText, isDeleting, phraseIndex, phrases, typingSpeed, deletingSpeed, pauseTime]);
    92→
    93→  return displayText;
    94→}

    96→/* ═══════════════════════════════════════════════════════════════════════
    97→   COUNTER ANIMATION HOOK

    99→
   100→function useCountUp(end: number, duration = 2000, startOnView = true) {
   101→  const [count, setCount] = useState(0);
   102→  const [started, setStarted] = useState(!startOnView);
   103→  const ref = useRef<HTMLDivElement>(null);
   104→
   105→  useEffect(() => {
   106→    if (!startOnView) return;
   107→    const el = ref.current;
   108→    if (!el) return;
   109→    const observer = new IntersectionObserver(
   110→      ([entry]) => {
   111→        if (entry.isIntersecting) {
   112→          setStarted(true);
   113→          observer.disconnect();
   114→        }
   115→      },
   116→      { threshold: 0.3 },
   117→    );
   118→    observer.observe(el);
   119→    return () => observer.disconnect();
   120→  }, [startOnView]);
   121→
   122→  useEffect(() => {
   123→    if (!started) return;
   124→    let startTime: number | null = null;
   125→    const step = (timestamp: number) => {
   126→      if (!startTime) startTime = timestamp;
   127→      const progress = Math.min((timestamp - startTime) / duration, 1);
   128→      const eased = 1 - Math.pow(1 - progress, 3);
   129→      setCount(Math.floor(eased * end));
   130→      if (progress < 1) requestAnimationFrame(step);
   131→    };
   132→    requestAnimationFrame(step);
   133→  }, [started, end, duration]);
   134→
   135→  return { count, ref };
   136→}
