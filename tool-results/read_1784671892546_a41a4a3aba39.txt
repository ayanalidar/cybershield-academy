   248→function Particles() {
   249→  const [particles, setParticles] = useState<Array<{id:number;left:string;delay:string;duration:string;size:string;color:string}>>([]);
   250→
   251→  useEffect(() => {
   252→    setParticles(
   253→      Array.from({ length: 30 }, (_, i) => ({
   254→        id: i,
   255→        left: `${Math.random() * 100}%`,
   256→        delay: `${Math.random() * 8}s`,
   257→        duration: `${8 + Math.random() * 12}s`,
   258→        size: `${1 + Math.random() * 2}px`,
   259→        color: i % 3 === 0 ? 'rgba(0,240,255,0.4)' : i % 3 === 1 ? 'rgba(191,0,255,0.3)' : 'rgba(57,255,20,0.3)',
   260→      }))
   261→    );
   262→  }, []);
   263→
   264→  return (
   265→    <div className="holo-bg">
   266→      {particles.map(p => (
   267→        <div
   268→          key={p.id}
   269→          className="particle"
   270→          style={{
   271→            left: p.left,
   272→            width: p.size,
   273→            height: p.size,
   274→            background: p.color,
   275→            animationDelay: p.delay,
   276→            animationDuration: p.duration,
   277→          }}
   278→        />
   279→      ))}
   280→    </div>
   281→  );
   282→}
