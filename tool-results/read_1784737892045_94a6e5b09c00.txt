   639→function QuizzesTab() {

   641→  const [currentQ, setCurrentQ] = useState(0);
   642→  const [selected, setSelected] = useState<string | null>(null);
   643→  const [answered, setAnswered] = useState(false);
   644→  const [score, setScore] = useState(0);
   645→  const [showHint, setShowHint] = useState(false);
   646→  const [timeLeft, setTimeLeft] = useState(DEMO_QUIZ.timeLimit);
   647→  const [finished, setFinished] = useState(false);

   651→  useEffect(() => {
   652→    if (!started || finished) return;
   653→    const timer = setInterval(() => { setTimeLeft(t => { if (t <= 1) { clearInterval(timer); setFinished(true); return 0; } return t - 1; }); }, 1000);
   654→    return () => clearInterval(timer);
   655→  }, [started, finished]);
   656→
   657→  const handleAnswer = useCallback((option: string) => {
   658→    if (answered) return;
   659→    setSelected(option);
   660→    setAnswered(true);
   661→    if (option === q.correct) setScore(s => s + q.points);
   662→  }, [answered, q]);
   663→
   664→  const nextQuestion = useCallback(() => {
   665→    if (currentQ < quiz.questions.length - 1) { setCurrentQ(c => c + 1); setSelected(null); setAnswered(false); setShowHint(false); }
   666→    else setFinished(true);
   667→  }, [currentQ, quiz.questions.length]);
   668→
   669→  if (!started) {
   670→    return (
   671→      <div className="max-w-lg mx-auto">
   672→        <div className="holo-card rounded-2xl p-8 text-center">

   677→          <Button onClick={() => setStarted(true)} className="holo-btn holo-btn-primary"><Play className="h-4 w-4 mr-2" />Start Quiz</Button>
   678→        </div>
   679→      </div>
   680→    );
   681→  }

   683→  if (finished) {
   684→    const pct = Math.round((score / (quiz.questions.reduce((a, q) => a + q.points, 0))) * 100);
   685→    return (
   686→      <div className="max-w-lg mx-auto">
   687→        <div className="holo-card rounded-2xl p-8 text-center neon-border-animated">
