   137→
   138→function MatrixRain() {
   139→  const columns = useMemo(() => {
   140→    const cols = [];
   141→    const count = typeof window !== 'undefined' ? Math.floor(window.innerWidth / 28) : 35;
   142→    for (let i = 0; i < count; i++) {
   143→      const chars = [];
   144→      const len = 15 + Math.floor(Math.random() * 25);
   145→      for (let j = 0; j < len; j++) {
   146→        chars.push(MATRIX_CHARS[Math.floor(Math.random() * MATRIX_CHARS.length)]);
   147→      }
   148→      cols.push({
   149→        left: `${(i / count) * 100}%`,
   150→        chars: chars.join(''),
   151→        duration: 8 + Math.random() * 15,
   152→        delay: Math.random() * -20,
   153→        opacity: 0.04 + Math.random() * 0.12,
   154→        fontSize: 10 + Math.floor(Math.random() * 8),
   155→      });
   156→    }
   157→    return cols;
   158→  }, []);
   159→
   160→  return (
   161→    <div className="matrix-rain">
   162→      {columns.map((col, i) => (
   163→        <div
   164→          key={i}
   165→          className="matrix-column"
   166→          style={{
   167→            left: col.left,
   168→            animationDuration: `${col.duration}s`,
   169→            animationDelay: `${col.delay}s`,
   170→            opacity: col.opacity,
   171→            fontSize: `${col.fontSize}px`,
   172→          }}
   173→        >
   174→          {col.chars}
   175→        </div>
   176→      ))}
   177→    </div>
   178→  );
   179→}

   182→   ANIMATED TERMINAL

   184→
   185→function AnimatedTerminal() {
   186→  const [visibleLines, setVisibleLines] = useState(0);
   187→
   188→  useEffect(() => {
   189→    const timers: NodeJS.Timeout[] = [];
   190→    TERMINAL_LINES.forEach((line, i) => {
   191→      timers.push(
   192→        setTimeout(() => {
   193→          setVisibleLines(i + 1);
   194→        }, line.delay + 1500)
   195→      );
   196→    });
   197→    // Loop: reset after all lines shown
   198→    const totalDelay = TERMINAL_LINES[TERMINAL_LINES.length - 1].delay + 4000;
   199→    const loopTimer = setInterval(() => {
   200→      setVisibleLines(0);
   201→      setTimeout(() => {
   202→        TERMINAL_LINES.forEach((line, i) => {
   203→          timers.push(
   204→            setTimeout(() => setVisibleLines(i + 1), line.delay)
   205→          );
   206→        });
   207→      }, 300);
   208→    }, totalDelay);
   209→    return () => {
   210→      timers.forEach(clearTimeout);
   211→      clearInterval(loopTimer);
   212→    };
   213→  }, []);
   214→
   215→  return (
   216→    <div className="rounded-xl overflow-hidden neon-border bg-[#0a0e17]/90 dark:bg-[#060a12]/95 backdrop-blur-md">
   217→      {/* Terminal title bar */}
   218→      <div className="flex items-center gap-2 px-4 py-2.5 border-b border-[#00ff88]/10 bg-[#00ff88]/5">
   219→        <div className="flex gap-1.5">
   220→          <div className="w-3 h-3 rounded-full bg-[#ff0040]/80" />
   221→          <div className="w-3 h-3 rounded-full bg-[#fbbf24]/80" />
   222→          <div className="w-3 h-3 rounded-full bg-[#00ff88]/80" />
   223→        </div>
   224→        <span className="text-xs text-[#00ff88]/60 font-mono ml-2">student@cybershield-lab:~</span>
   225→      </div>
   226→      {/* Terminal body */}
   227→      <div className="p-4 font-mono text-xs leading-relaxed min-h-[200px]">
   228→        {TERMINAL_LINES.slice(0, visibleLines).map((line, i) => (
   229→          <motion.div
   230→            key={`${i}-${visibleLines}`}
   231→            initial={{ opacity: 0, x: -10 }}
   232→            animate={{ opacity: 1, x: 0 }}
   233→            transition={{ duration: 0.15 }}
   234→            className={`${line.color} whitespace-pre`}
   235→          >
   236→            {line.text || '\u00A0'}
   237→          </motion.div>
   238→        ))}
   239→        {visibleLines > 0 && (
   240→          <span className="terminal-cursor" />
   241→        )}
   242→      </div>
   243→    </div>
   244→  );
   245→}
