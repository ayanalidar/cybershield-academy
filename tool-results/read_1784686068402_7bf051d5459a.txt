   107→function resolvePath(cwd: string, target: string): string {
   108→  if (target.startsWith('/')) return normalizePath(target);
   109→  if (target === '..') { const parts = cwd.split('/').filter(Boolean); parts.pop(); return '/' + parts.join('/'); }
   110→  if (target === '.') return cwd;
   111→  return normalizePath(cwd + '/' + target);
   112→}

   114→function normalizePath(p: string): string {
   115→  const parts = p.split('/').filter(Boolean);
   116→  const resolved: string[] = [];
   117→  for (const part of parts) { if (part === '..') resolved.pop(); else if (part !== '.') resolved.push(part); }
   118→  return '/' + resolved.join('/');

   839→function checkObjectives(session: TerminalSession, command: string): void {
   840→  const lowerCmd = command.toLowerCase();
   841→  for (const obj of session.objectives) {
   842→    if (obj.completed) continue;
   843→    if (obj.verificationPattern) {
   844→      if (new RegExp(obj.verificationPattern, 'i').test(lowerCmd)) { obj.completed = true; continue; }
   845→    }
   846→    for (const [keyword, regex] of Object.entries(LAB_OBJECTIVE_PATTERNS)) {
   847→      if (obj.description.toLowerCase().includes(keyword) && regex.test(lowerCmd)) { obj.completed = true; break; }
   848→    }
   849→  }
   850→  const completed = session.objectives.filter(o => o.completed).length;
   851→  if (completed === session.objectives.length && session.objectives.length > 0) {
   852→    io.to(session.labSessionId).emit('lab:completed', { labSessionId: session.labSessionId, message: 'All lab objectives completed! Great work.', score: 1.0 });
   853→  }
   854→}
   855→
   856→io.on('connection', (socket) => {
   857→  console.log(`[Terminal] Client connected: ${socket.id}`);
   858→
   859→  socket.on('terminal:join', (data: { userId: string; labSessionId: string; containerName: string; objectives: { id: string; description: string; verificationPattern?: string }[]; labType?: string }) => {
   860→    const { userId, labSessionId, containerName, objectives, labType } = data;
   861→    const session: TerminalSession = {

   865→      labType: labType || 'general',
   866→      objectives: objectives.map(o => ({ ...o, completed: false })),
   867→      startedAt: new Date(),
   868→    };
   869→    activeSessions.set(socket.id, session);
   870→    socket.join(labSessionId);
   871→
   872→    const welcomeLines = [

   903→      const prompt = `\x1b[1;32mstudent@cybershield\x1b[0m:\x1b[1;34m${session.cwd}\x1b[0m$ `;
