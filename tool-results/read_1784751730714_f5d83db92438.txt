   415→   415→function ProfessorTab() {
   416→   416→  const [messages, setMessages] = useState<ChatMsg[]>([]);
   417→   417→  const [input, setInput] = useState('');
   418→   418→  const [voiceState, setVoiceState] = useState<VoiceState>('idle');
   419→   419→  const [selectedVoice, setSelectedVoice] = useState(VOICE_PROFILES[0]);
   420→   420→  const [voiceSettingsOpen, setVoiceSettingsOpen] = useState(false);
   421→   421→  const [isMuted, setIsMuted] = useState(false);
   422→   422→  const [streamingText, setStreamingText] = useState('');
   423→   423→
   424→   424→  // Refs to avoid stale closures
   425→   425→  const messagesRef = useRef(messages);
   426→   426→  const voiceStateRef = useRef(voiceState);
   427→   427→  const isProcessingRef = useRef(false);
   428→   428→  const audioRef = useRef<HTMLAudioElement | null>(null);
   429→   429→  const mediaRecorderRef = useRef<MediaRecorder | null>(null);
   430→   430→  const streamRef = useRef<MediaStream | null>(null);
   431→   431→  const chatEndRef = useRef<HTMLDivElement>(null);
   432→   432→  const inputRef = useRef<HTMLInputElement>(null);
   433→   433→  const stopSpeakingRef = useRef<(() => void) | null>(null);
   434→   434→
   435→   435→  useEffect(() => { messagesRef.current = messages; }, [messages]);
   436→   436→  useEffect(() => { voiceStateRef.current = voiceState; }, [voiceState]);
   437→   437→
   438→   438→  // Auto-scroll chat
   439→   439→  useEffect(() => {
   440→   440→    chatEndRef.current?.scrollIntoView({ behavior: 'smooth' });
   441→   441→  }, [messages, streamingText]);
   442→   442→

   444→   444→  const stopSpeaking = useCallback(() => {
   445→   445→    if (audioRef.current) {
   446→   446→      audioRef.current.pause();
   447→   447→      audioRef.current.currentTime = 0;
   448→   448→      audioRef.current = null;
   449→   449→    }
   450→   450→    if (stopSpeakingRef.current) {
   451→   451→      stopSpeakingRef.current();
   452→   452→      stopSpeakingRef.current = null;
   453→   453→    }
   454→   454→    if (voiceStateRef.current === 'speaking') {
   455→   455→      setVoiceState('idle');
   456→   456→    }
   457→   457→  }, []);
   458→   458→
   459→   459→  // Speak text using TTS
   460→   460→  const speakText = useCallback(async (text: string) => {
   461→   461→    if (isMuted) return;
   462→   462→    try {
   463→   463→      setVoiceState('speaking');
   464→   464→      const res = await fetch('/api/voice/tts', {
   465→   465→        method: 'POST',
   466→   466→        headers: { 'Content-Type': 'application/json' },
   467→   467→        body: JSON.stringify({
   468→   468→          text: text.slice(0, 1024),
   469→   469→          voice: selectedVoice.id,
   470→   470→          speed: selectedVoice.speed,
   471→   471→        }),
   472→   472→      });
   473→   473→
   474→   474→      if (!res.ok) {
   475→   475→        console.error('TTS failed:', res.status);
   476→   476→        setVoiceState('idle');
   477→   477→        return;
   478→   478→      }
   479→   479→
   480→   480→      const blob = await res.blob();
   481→   481→      const url = URL.createObjectURL(blob);
   482→   482→      const audio = new Audio(url);
   483→   483→      audioRef.current = audio;
   484→   484→
   485→   485→      const onEnded = () => {
   486→   486→        URL.revokeObjectURL(url);
   487→   487→        audioRef.current = null;
   488→   488→        stopSpeakingRef.current = null;
   489→   489→        setVoiceState('idle');
   490→   490→      };
   491→   491→
   492→   492→      const onError = () => {
   493→   493→        URL.revokeObjectURL(url);
   494→   494→        audioRef.current = null;
   495→   495→        stopSpeakingRef.current = null;
   496→   496→        setVoiceState('idle');
   497→   497→      };
   498→   498→
   499→   499→      audio.addEventListener('ended', onEnded, { once: true });
   500→   500→      audio.addEventListener('error', onError, { once: true });
   501→   501→      stopSpeakingRef.current = () => {
   502→   502→        audio.removeEventListener('ended', onEnded);
   503→   503→        audio.removeEventListener('error', onError);
   504→   504→        onEnded();
   505→   505→      };
   506→   506→
   507→   507→      await audio.play();
   508→   508→    } catch (err) {
   509→   509→      console.error('Speech error:', err);
   510→   510→      setVoiceState('idle');
   511→   511→    }
   512→   512→  }, [selectedVoice, isMuted]);
   513→   513→

   515→   515→  const sendMessage = useCallback(async (text: string) => {
   516→   516→    if (!text.trim() || isProcessingRef.current) return;

   519→   519→    const userMsg: ChatMsg = { role: 'user', content: text.trim() };
   520→   520→    const newMessages = [...messagesRef.current, userMsg];
   521→   521→    setMessages(newMessages);
   522→   522→    setInput('');
   523→   523→    setStreamingText('');
   524→   524→
   525→   525→    try {
   526→   526→      const history = newMessages.slice(-10).map(m => ({ role: m.role, content: m.content }));
   527→   527→
   528→   528→      const res = await fetch('/api/chat', {
   529→   529→        method: 'POST',
   530→   530→        headers: { 'Content-Type': 'application/json' },
   531→   531→        body: JSON.stringify({
   532→   532→          userId: DEMO_USER.id,
   533→   533→          sessionId: 'academy-session',
   534→   534→          message: text.trim(),
   535→   535→          history: history.slice(0, -1),
   536→   536→        }),
   537→   537→      });
   538→   538→
   539→   539→      if (!res.ok) {
   540→   540→        const errData = await res.json().catch(() => null);
   541→   541→        const errMsg = errData?.error || 'Something went wrong. Please try again.';
   542→   542→        setMessages(prev => [...prev, { role: 'assistant', content: errMsg }]);
   543→   543→        isProcessingRef.current = false;
   544→   544→        return;
   545→   545→      }
   546→   546→
   547→   547→      // Stream the response
   548→   548→      const reader = res.body?.getReader();
   549→   549→      if (!reader) {
   550→   550→        setMessages(prev => [...prev, { role: 'assistant', content: 'Failed to read response.' }]);
   551→   551→        isProcessingRef.current = false;
   552→   552→        return;
   553→   553→      }
   554→   554→
   555→   555→      const decoder = new TextDecoder();
   556→   556→      let fullText = '';

   559→   559→        const { done, value } = await reader.read();
   560→   560→        if (done) break;
   561→   561→        const chunk = decoder.decode(value, { stream: true });
   562→   562→        fullText += chunk;
   563→   563→        setStreamingText(fullText);
   564→   564→      }
   565→   565→
   566→   566→      setStreamingText('');
   567→   567→      setMessages(prev => [...prev, { role: 'assistant', content: fullText }]);
   568→   568→
   569→   569→      // Speak the response

   571→   571→        speakText(fullText);
   572→   572→      }
   573→   573→    } catch (err) {
   574→   574→      console.error('Chat error:', err);
   575→   575→      setMessages(prev => [...prev, { role: 'assistant', content: 'Network error. Please check your connection and try again.' }]);
   576→   576→    } finally {
   577→   577→      isProcessingRef.current = false;
   578→   578→    }
   579→   579→  }, [speakText, isMuted]);
   580→   580→

   621→   621→          }
   622→   622→        } catch {
