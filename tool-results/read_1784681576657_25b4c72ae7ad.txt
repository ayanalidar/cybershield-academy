    61→    61→function useTypingEffect(phrases: string[], typingSpeed = 80, deletingSpeed = 40, pauseTime = 2000) {
    62→    62→  const [displayText, setDisplayText] = useState('');
    63→    63→  const [phraseIndex, setPhraseIndex] = useState(0);
    64→    64→  const [isDeleting, setIsDeleting] = useState(false);
    65→    65→
    66→    66→  useEffect(() => {
    67→    67→    const currentPhrase = phrases[phraseIndex];
    68→    68→    let timeout: NodeJS.Timeout;
    69→    69→
    70→    70→    if (!isDeleting && displayText === currentPhrase) {
    71→    71→      timeout = setTimeout(() => setIsDeleting(true), pauseTime);
    72→    72→    } else if (isDeleting && displayText === '') {
    73→    73→      timeout = setTimeout(() => {
    74→    74→        setIsDeleting(false);
    75→    75→        setPhraseIndex((prev) => (prev + 1) % phrases.length);
    76→    76→      }, 0);
    77→    77→    } else {
    78→    78→      timeout = setTimeout(
    79→    79→        () => {
    80→    80→          setDisplayText(
    81→    81→            isDeleting
    82→    82→              ? currentPhrase.substring(0, displayText.length - 1)
    83→    83→              : currentPhrase.substring(0, displayText.length + 1),
    84→    84→          );
    85→    85→        },
    86→    86→        isDeleting ? deletingSpeed : typingSpeed,
    87→    87→      );
    88→    88→    }
    89→    89→
    90→    90→    return () => clearTimeout(timeout);
    91→    91→  }, [displayText, isDeleting, phraseIndex, phrases, typingSpeed, deletingSpeed, pauseTime]);
    92→    92→
    93→    93→  return displayText;
    94→    94→}

    99→    99→
   100→   100→function useCountUp(end: number, duration = 2000, startOnView = true) {
   101→   101→  const [count, setCount] = useState(0);
   102→   102→  const [started, setStarted] = useState(!startOnView);
   103→   103→  const ref = useRef<HTMLDivElement>(null);
   104→   104→
   105→   105→  useEffect(() => {
   106→   106→    if (!startOnView) return;
   107→   107→    const el = ref.current;
   108→   108→    if (!el) return;
   109→   109→    const observer = new IntersectionObserver(
   110→   110→      ([entry]) => {
   111→   111→        if (entry.isIntersecting) {
   112→   112→          setStarted(true);
   113→   113→          observer.disconnect();
   114→   114→        }
   115→   115→      },
   116→   116→      { threshold: 0.3 },
   117→   117→    );
   118→   118→    observer.observe(el);
   119→   119→    return () => observer.disconnect();
   120→   120→  }, [startOnView]);
   121→   121→
   122→   122→  useEffect(() => {
   123→   123→    if (!started) return;
   124→   124→    let startTime: number | null = null;
   125→   125→    const step = (timestamp: number) => {
   126→   126→      if (!startTime) startTime = timestamp;
   127→   127→      const progress = Math.min((timestamp - startTime) / duration, 1);
   128→   128→      const eased = 1 - Math.pow(1 - progress, 3);
   129→   129→      setCount(Math.floor(eased * end));
   130→   130→      if (progress < 1) requestAnimationFrame(step);
   131→   131→    };
   132→   132→    requestAnimationFrame(step);
   133→   133→  }, [started, end, duration]);
   134→   134→
   135→   135→  return { count, ref };
   136→   136→}
