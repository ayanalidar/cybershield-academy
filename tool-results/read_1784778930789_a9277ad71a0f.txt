   137→function MatrixRain() {
   138→  const columns = useMemo(() => {
   139→    const cols = [];
   140→    const count = typeof window !== 'undefined' ? Math.floor(window.innerWidth / 28) : 35;
   141→    for (let i = 0; i < count; i++) {
   142→      const chars = [];
   143→      const len = 15 + Math.floor(Math.random() * 25);
   144→      for (let j = 0; j < len; j++) {
   145→        chars.push(MATRIX_CHARS[Math.floor(Math.random() * MATRIX_CHARS.length)]);
   146→      }
   147→      cols.push({
   148→        left: `${(i / count) * 100}%`,
   149→        chars: chars.join(''),
   150→        duration: 8 + Math.random() * 15,
   151→        delay: Math.random() * -20,
   152→        opacity: 0.04 + Math.random() * 0.12,
   153→        fontSize: 10 + Math.floor(Math.random() * 8),
   154→      });
   155→    }
   156→    return cols;
   157→  }, []);
   158→
   159→  return (
   160→    <div className="matrix-rain">
   161→      {columns.map((col, i) => (
   162→        <div
   163→          key={i}
   164→          className="matrix-column"
   165→          style={{
   166→            left: col.left,
   167→            animationDuration: `${col.duration}s`,
   168→            animationDelay: `${col.delay}s`,
   169→            opacity: col.opacity,
   170→            fontSize: `${col.fontSize}px`,
   171→          }}
   172→        >
   173→          {col.chars}
   174→        </div>
   175→      ))}
   176→    </div>
   177→  );
   178→}

   183→
   184→function AnimatedTerminal() {
   185→  const [visibleLines, setVisibleLines] = useState(0);
   186→
   187→  useEffect(() => {
   188→    const timers: NodeJS.Timeout[] = [];
   189→    TERMINAL_LINES.forEach((line, i) => {
   190→      timers.push(
   191→        setTimeout(() => {
   192→          setVisibleLines(i + 1);
   193→        }, line.delay + 1500)
   194→      );
   195→    });
   196→    // Loop: reset after all lines shown
   197→    const totalDelay = TERMINAL_LINES[TERMINAL_LINES.length - 1].delay + 4000;
   198→    const loopTimer = setInterval(() => {
   199→      setVisibleLines(0);
   200→      setTimeout(() => {
   201→        TERMINAL_LINES.forEach((line, i) => {
   202→          timers.push(
   203→            setTimeout(() => setVisibleLines(i + 1), line.delay)
   204→          );
   205→        });
   206→      }, 300);
   207→    }, totalDelay);
   208→    return () => {
   209→      timers.forEach(clearTimeout);
   210→      clearInterval(loopTimer);
   211→    };
   212→  }, []);
   213→
   214→  return (
   215→    <div className="rounded-xl overflow-hidden neon-border bg-[#0a0e17]/90 dark:bg-[#060a12]/95 backdrop-blur-md">
   216→      {/* Terminal title bar */}
   217→      <div className="flex items-center gap-2 px-4 py-2.5 border-b border-[#00ff88]/10 bg-[#00ff88]/5">
   218→        <div className="flex gap-1.5">
   219→          <div className="w-3 h-3 rounded-full bg-[#ff0040]/80" />
   220→          <div className="w-3 h-3 rounded-full bg-[#fbbf24]/80" />
   221→          <div className="w-3 h-3 rounded-full bg-[#00ff88]/80" />
   222→        </div>
   223→        <span className="text-xs text-[#00ff88]/60 font-mono ml-2">student@cybershield-lab:~</span>
   224→      </div>
   225→      {/* Terminal body */}
   226→      <div className="p-4 font-mono text-xs leading-relaxed min-h-[200px]">
   227→        {TERMINAL_LINES.slice(0, visibleLines).map((line, i) => (
   228→          <motion.div
   229→            key={`${i}-${visibleLines}`}
   230→            initial={{ opacity: 0, x: -10 }}
   231→            animate={{ opacity: 1, x: 0 }}
   232→            transition={{ duration: 0.15 }}
   233→            className={`${line.color} whitespace-pre`}
   234→          >
   235→            {line.text || '\u00A0'}
   236→          </motion.div>
   237→        ))}
   238→        {visibleLines > 0 && (
   239→          <span className="terminal-cursor" />
   240→        )}
   241→      </div>

   243→  );
   244→}
