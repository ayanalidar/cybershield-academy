   261→function formatTime(isoString: string): string {
   262→  const d = new Date(isoString);
   263→  return d.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });

   266→function formatDate(isoString: string): string {
   267→  const d = new Date(isoString);
   268→  return d.toLocaleDateString('en-US', {
   269→    year: 'numeric',
   270→    month: 'long',
   271→    day: 'numeric',
   272→  });
   273→}

   275→function getFocusColor(score: number): string {
   276→  if (score >= 75) return 'bg-emerald-500 text-white';
   277→  if (score >= 50) return 'bg-yellow-500 text-white';
   278→  return 'bg-red-500 text-white';
   279→}

   282→  if (score >= 75) return 'Focused';
